2
ตั้งค่า Google Form ทุกอัน — สำคัญมาก
🔴 บังคับทำทุก Form โดยไม่มีข้อยกเว้น ถ้าไม่ตั้งค่านี้ ระบบจะไม่มีการตรวจสอบ email จาก Google และจะไม่ทราบว่าใครเป็นผู้ทำแบบทดสอบ

เปิด Google Form → คลิกแท็บ Settings (⚙️)

ไปที่หัวข้อ Responses

หา "Collect email addresses" → เลือก "Verified"

⚠️ ต้องเลือก Verified ไม่ใช่ Responder input — Verified คือให้ Google ตรวจสอบ Account จริง

บันทึกการตั้งค่า

ทำซ้ำกับทุก Form ที่ต้องการใช้งานในระบบนี้

✅ ผลลัพธ์ที่ได้ Google จะบังคับให้ผู้ใช้ Login ด้วย Google Account จริงก่อนกรอก Form email ที่ได้จึงยืนยันตัวตนได้ 100% — ปลอมแปลงไม่ได้
3
ติดตั้ง Apps Script ใน LoginSheet

เปิด LoginSheet → เมนู Extensions → Apps Script

ไฟล์ Code.gs ที่มีอยู่ → ลบโค้ดเดิมทั้งหมด → วางโค้ดด้านล่าง

เพิ่มไฟล์ใหม่ → คลิก + → เลือก HTML → ตั้งชื่อ IndexPage → วางโค้ด HTML ด้านล่าง

ชื่อต้องเป็น IndexPage ตรงๆ ไม่มี .html ต่อท้าย

บันทึกทั้ง 2 ไฟล์ → Ctrl+S

Code.gs Google Apps Script 
// ===================================================================
// Owlvy House — Form Selector (Code.gs)
// วางใน Apps Script ของ LoginSheet
// Deploy → Execute as: Me | Who has access: Anyone
// ===================================================================

const SHEET_FORMS = "Forms";

const F_CODE   = 1;
const F_TITLE  = 2;
const F_URL    = 3;
const F_STATUS = 4;

// ===================================================================
// doGet — serve หน้าเลือกชุดแบบฝึกหัด
// ===================================================================
function doGet(e) {
  const template = HtmlService.createTemplateFromFile("IndexPage");
  template.forms = getPublishedForms();
  return template
    .evaluate()
    .setTitle("Owlvy House — เลือกชุดแบบฝึกหัด")
    .addMetaTag("viewport", "width=device-width, initial-scale=1");
}

// ===================================================================
// getPublishedForms — อ่าน Sheet Forms เฉพาะ Status = publish
// ===================================================================
function getPublishedForms() {
  try {
    const ss        = SpreadsheetApp.getActiveSpreadsheet();
    const formSheet = ss.getSheetByName(SHEET_FORMS);
    const data      = formSheet.getDataRange().getValues();
    const result    = [];

    for (let i = 1; i < data.length; i++) {
      const code   = String(data[i][F_CODE   - 1]).trim();
      const title  = String(data[i][F_TITLE  - 1]).trim();
      const rawUrl = String(data[i][F_URL    - 1]).trim();
      const status = String(data[i][F_STATUS - 1]).trim().toLowerCase();

      if (!code || !rawUrl || status !== "publish") continue;

      // รองรับทั้ง ID เปล่า และ URL เต็ม
      const url = rawUrl.startsWith("http")
        ? rawUrl
        : "https://docs.google.com/forms/d/" + rawUrl + "/viewform";

      result.push({ code: code, title: title, url: url });
    }

    return result;

  } catch (err) {
    Logger.log("getPublishedForms error: " + err.message);
    return [];
  }
}

Deploy เป็น Web App

คลิกปุ่ม Deploy มุมบนขวา → New deployment

คลิกไอคอน ⚙️ → เลือก Web app

ตั้งค่าดังนี้:

ตัวเลือกค่าที่ต้องเลือก
Execute asMe
Who has accessAnyone

คลิก Deploy → ขอสิทธิ์ → Authorize access → Allow

ถ้าขึ้น "Google hasn't verified" → คลิก Advanced → Go to (unsafe) → Allow

คัดลอก Web app URL → แชร์ให้ผู้ใช้ได้เลย

⚠️ ถ้าแก้โค้ดหลัง Deploy แล้ว ต้อง Deploy ใหม่ทุกครั้ง: Deploy → Manage deployments → ✏️ Edit → Version: New version → Deploy
5
การจัดการระบบหลัง Deploy
ต้องการวิธีทำ
เปิด Form ใหม่ให้ผู้ใช้เห็น เปลี่ยน Status ใน Sheet เป็น publish
ปิด Form ชั่วคราว เปลี่ยน Status เป็น waiting หรืออื่นๆ
เพิ่ม Form ใหม่ เพิ่มแถวใน Sheet Forms ใส่ FormCode, FormTitle, FormURL (ID), Status
ดูว่าใครทำแบบทดสอบ เปิด Google Form → Responses — มี email ทุกคนบันทึกไว้
