Firebase Script ที่ต้องใส่

<script type="module">
// ── IMPORTS ──────────────────────────
import { initializeApp } 
  from "https://www.gstatic.com/firebasejs/12.0.0/firebase-app.js";
import { getAuth, GoogleAuthProvider, 
         signInWithPopup, onAuthStateChanged,
         signOut as fbSignOut } 
  from "https://www.gstatic.com/firebasejs/12.0.0/firebase-auth.js";
import { getFirestore } 
  from "https://www.gstatic.com/firebasejs/12.0.0/firebase-firestore.js";

// ── CONFIG (แก้ตรงนี้เท่านั้น!) ──────
const firebaseConfig = {
  apiKey: "YOUR_API_KEY",
  authDomain: "YOUR_PROJECT.firebaseapp.com",
  projectId: "YOUR_PROJECT",
  storageBucket: "YOUR_PROJECT.firebasestorage.app",
  messagingSenderId: "YOUR_SENDER_ID",
  appId: "YOUR_APP_ID"
};

// ── INIT ──────────────────────────────
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const provider = new GoogleAuthProvider();

// ── AUTH STATE ────────────────────────
onAuthStateChanged(auth, (user) => {
  if (user) {
    // Login แล้ว → แสดงแอป
    document.getElementById('login-page').style.display = 'none';
    document.getElementById('app').style.display = 'block';
    document.getElementById('user-name').textContent = 
      user.displayName || user.email;
  } else {
    // ยังไม่ Login → แสดงหน้า Login
    document.getElementById('login-page').style.display = 'flex';
    document.getElementById('app').style.display = 'none';
  }
});

// ── FUNCTIONS ─────────────────────────
window.signInWithGoogle = async () => {
  try {
    await signInWithPopup(auth, provider);
  } catch(e) {
    alert('Login ไม่สำเร็จ: ' + e.message);
  }
};

window.signOut = async () => {
  if (confirm('ต้องการออกจากระบบ?')) {
    await fbSignOut(auth);
  }
};
</script>

onAuthStateChanged(auth, (user) => {
  if (user) {
    console.log('UID:', user.uid);
    console.log('ชื่อ:', user.displayName);
    console.log('Email:', user.email);
    console.log('รูปโปรไฟล์:', user.photoURL);

    console.log('Login ล่าสุด:', user.metadata.lastSignInTime);
  }
});

บันทึก Log การ Login ลง Firestore

ทำแบบนี้เพื่อดูประวัติการ Login ย้อนหลังได้
import { getFirestore, doc, setDoc, 
         serverTimestamp } 
  from "firebase/firestore";

// บันทึกทุกครั้งที่ Login
onAuthStateChanged(auth, async (user) => {
  if (user) {
    // บันทึกลง Firestore
    await setDoc(
      doc(db, 'users', user.uid), {
        name: user.displayName,
        email: user.email,
        photo: user.photoURL,
        lastLogin: serverTimestamp(),
        loginCount: increment(1)
      }, { merge: true }
    );
  }
});
